144  /// <summary>
145  /// Gets the actual exchange the index lives on
146  /// </summary>
147  /// <remarks>Useful for live trading</remarks>
148  /// <returns>The exchange of the index</returns>
149  public static string GetIndexExchange(Symbol symbol)
150  {
151  return _indexExchange.TryGetValue(symbol.Value, out var market)
152  ? market
153  : symbol.ID.Market;
154  }

156  /// <summary>

158  /// </summary>
159  /// <returns>The market of the index</returns>
160  public static bool TryGetIndexMarket(string ticker, out string market)
161  {
